Quick Start

All components use Tailwind CSS via CDN. Add this to your HTML:

<script src="https://cdn.tailwindcss.com"></script>
<script>
    tailwind.config = {
        theme: {
            extend: {
                colors: {
                    primary: {
                        50: '#faf5ff',
                        600: '#9333ea',
                        700: '#7e22ce',
                        800: '#6b21a8',
                        900: '#581c87',
                    }
                }
            }
        }
    }
</script>
